+def _cond_matches(actual, cond):
+    if not isinstance(cond, dict):
+        return actual == cond
+    for op, expected in cond.items():
+        if op == "eq":
+            if not (actual == expected):
+                return False
+        elif op in ("ne", "neq", "not_eq"):
+            if not (actual != expected):
+                return False
+        elif op == "in":
+            if not (actual in expected):
+                return False
+        elif op in ("nin", "not_in"):
+            if not (actual not in expected):
+                return False
+        elif op == "gt":
+            if not (actual is not None and actual > expected):
+                return False
+        elif op == "gte":
+            if not (actual is not None and actual >= expected):
+                return False
+        elif op == "lt":
+            if not (actual is not None and actual < expected):
+                return False
+        elif op == "lte":
+            if not (actual is not None and actual <= expected):
+                return False
+        else:
+            # Unknown operator: fall back to equality against the condition itself
+            return False
+    return True
+
+def _rule_matches(rule, input_dict):
+    when = rule.get("when", {}) or {}
+    for field, cond in when.items():
+        if not _cond_matches(input_dict.get(field), cond):
+            return False
+    return True
+
+def _ordered_rules(spec):
+    rules = list(spec.get("rules", []))
+    priority = spec.get("priority")
+    if priority:
+        rank = {rid: i for i, rid in enumerate(priority)}
+        # Rules in priority list first (in priority order), others after in original order
+        return sorted(rules, key=lambda r: (0, rank[r["id"]]) if r.get("id") in rank else (1, 0))
+    # Also honor decision.winner as precedence if priority absent but decision present
+    decision = spec.get("decision") or {}
+    winner = decision.get("winner")
+    if winner:
+        return sorted(rules, key=lambda r: (0 if r.get("id") == winner else 1))
+    return rules
